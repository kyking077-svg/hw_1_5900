When SAT scores are standardized, a one–standard deviation increase in a school’s average SAT score is associated with an approximately 15 percentage-point increase in six-year completion rates (p < 0.001). One standard deviation is about 150 SAT score points.

Sat and admissions rates scatterplot

quad_fit <- lm(ADM_RATE ~ SAT_AVG + I(SAT_AVG^2), data = admin_df)

#generating predicitons
x_seq <- seq(min(admin_df$SAT_AVG, na.rm = TRUE),
             max(admin_df$SAT_AVG, na.rm = TRUE),
             length.out = 200)

pred_df <- data.frame(SAT_AVG = x_seq)
pred <- predict(quad_fit, newdata = pred_df, interval = "confidence", level = 0.95)
pred_df$fit <- pred[, "fit"]
pred_df$lwr <- pred[, "lwr"]
pred_df$upr <- pred[, "upr"]

#chart
fig <- plot_ly() |>
  # Scatter points
  add_trace(data = admin_df,
            x = ~SAT_AVG, y = ~ADM_RATE,
            type = 'scatter', mode = 'markers',
            marker = list(color = sat_blue),
            name = 'Schools', text = ~paste(
    "School: ", INSTNM,
    "<br>Average SAT: ", SAT_AVG,
    "<br>Admission Rate: ", scales::percent(ADM_RATE, accuracy = 1)
  ),
  hoverinfo = 'text'
) |>
  # Regression line
  add_trace(data = pred_df,
            x = ~SAT_AVG, y = ~fit,
            type = 'scatter', mode = 'lines',
            line = list(color = sat_grey, width = 2),
            name = 'Line of Best Fit') |>
  # Confidence ribbon
  add_ribbons(data = pred_df,
              x = ~SAT_AVG,
              ymin = ~lwr, ymax = ~upr,
              fillcolor = 'rgba(70,130,180,0.2)',  # semi-transparent steelblue
              line = list(color = 'transparent'),
              name = '95% CI') |>
  layout(title = list(text="Relationship Between SAT Average Scores and Admission Rates",tickfont = list(size = 16, color = "black", family = "Arial")),
         xaxis = list(title = "SAT Average Score for Accepted Students", fixedrange = TRUE, tickfont = list(size = 14, color = "black", family = "Arial")),
         yaxis = list(title = "Admission Rate", fixedrange = TRUE, tickformat = ".0%", tickfont = list(size = 14, color = "black", family = "Arial")),
        dragmode = "zoom") |>
  config(
    modeBarButtonsToRemove = c(
      "select2d",   # box select
      "lasso2d",    # lasso select
      "toggleSpikelines"  # optional extra buttons you might not want
    ),
    displaylogo = FALSE  # removes Plotly logo
  )
